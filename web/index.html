AWS SAM Template (infra/template.yaml)
AWSTemplateFormatVersion: '2010-09-09'
Transform: AWS::Serverless-2016-10-31
Description: >
  Event-Driven Serverless AI Image Analyzer
  S3 ObjectCreated -> Lambda -> Rekognition -> DynamoDB (+DLQ)

Resources:
  # S3 Input Bucket
  SourceImageBucket:
    Type: AWS::S3::Bucket
    Properties:
      BucketName: !Sub "image-analyzer-${AWS::AccountId}-${AWS::Region}"
      NotificationConfiguration:
        LambdaConfigurations:
          - Event: "s3:ObjectCreated:*"
            Filter:
              S3Key:
                Rules:
                  - Name: prefix
                    Value: incoming/
            Function: !GetAtt ImageAnalyzerFunction.Arn

  # DynamoDB Table
  ImageTagsTable:
    Type: AWS::DynamoDB::Table
    Properties:
      TableName: ImageTags
      BillingMode: PAY_PER_REQUEST
      AttributeDefinitions:
        - AttributeName: ImageKey
          AttributeType: S
      KeySchema:
        - AttributeName: ImageKey
          KeyType: HASH
      SSESpecification:
        SSEEnabled: true

  # SQS Dead-Letter Queue (NEW in Review 3)
  ImageProcessingDLQ:
    Type: AWS::SQS::Queue
    Properties:
      QueueName: ImageProcessingDLQ
      MessageRetentionPeriod: 1209600  # 14 days
      VisibilityTimeout: 60

  # Lambda Function with DLQ config
  ImageAnalyzerFunction:
    Type: AWS::Serverless::Function
    Properties:
      FunctionName: ImageAnalyzerFunction
      CodeUri: ../lambda/
      Handler: handler.lambda_handler
      DeadLetterQueue:
        Type: SQS
        TargetArn: !GetAtt ImageProcessingDLQ.Arn
      Environment:
        Variables:
          DYNAMODB_TABLE_NAME: !Ref ImageTagsTable
          MIN_CONFIDENCE: "70.0"
          MAX_LABELS: "10"
          DLQ_URL: !Ref ImageProcessingDLQ
